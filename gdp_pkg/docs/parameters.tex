\subsection*{FSP Thermal Model — Heterogeneous Population}
% ─────────────────────────────────────────────────────────────────────────────

Parameters are drawn from truncated normal distributions
$\mathcal{TN}(\mu,\sigma^2; [\ell,u])$ independently for each FSP.
The thermal dynamics follow a first-order linear model:
$T^{\text{in}}_{i,t+1} = a_i T^{\text{in}}_{i,t} + b_i u_{i,t} + c_i \tilde{T}_{\text{out},t}$.

\begin{table}[htbp]
\centering
\caption{FSP thermal model parameters (heterogeneous population, $N=60$).}
\label{tab:fsp}
\begin{tabular}{llllll}
\toprule
Symbol & Parameter & $\mu$ & $[\ell,\, u]$ & Unit & Notes \\
\midrule
$a_i$           & Building thermal inertia          & 0.998 & $[0.994,\,0.999]$ & — & Dimensionless decay \\
$b_i$           & Heating efficiency                & 0.020 & $[0.010,\,0.035]$ & °C/kWh & $\sigma=0.004$ \\
$c_i$           & Outdoor coupling coefficient      & 0.002 & $[0.001,\,0.005]$ & °C/°C & $\sigma=0.001$ \\
$u_i^{\max}$    & Rated heating power               & 8.5   & $[6.5,\,13.0]$   & kW    & $\sigma=2.0$ \\

\subsection*{ADMM Algorithm Hyperparameters}
% ─────────────────────────────────────────────────────────────────────────────

\begin{table}[htbp]
\centering
\caption{Exchange ADMM hyperparameters (Boyd et al.~\cite{stephen_distributed_2011}, \S3.4.1 and \S7.3.2).}
\label{tab:admm}
\begin{tabular}{lllll}
\toprule
Symbol & Parameter & Value & Unit & Source \\
\midrule
$J$                  & Number of agents ($N$ FSPs + 1 aggregator) & 61 & — & $N+1$ \\
$\rho_0$             & Initial penalty parameter                  & 0.05  & — & \texttt{ADMMConfig.rho} \\
$\varepsilon^\text{pri}$ & Primal residual tolerance $\|\mathbf{r}_p\|_2 < \varepsilon^\text{pri}$ & $10^{-3}$ & — & \texttt{eps\_primal} \\
$\varepsilon^\text{dual}$ & Dual residual tolerance $\|\mathbf{r}_d\|_2 < \varepsilon^\text{dual}$ & $10^{-3}$ & — & \texttt{eps\_dual} \\
$n_\text{max}$       & Maximum iterations                         & 200   & — & \texttt{max\_iter} \\
$\mu$                & Residual ratio threshold for $\rho$ update & 8.0   & — & \texttt{mu\_res} \\
$\tau$               & $\rho$ increment/decrement factor           & 1.2   & — & \texttt{tau\_incr} \\
$\rho_\text{max}$    & Upper bound on $\rho$                      & 10.0  & — & \texttt{rho\_max} \\
$\rho_\text{min}$    & Lower bound on $\rho$                      & $10^{-4}$ & — & \texttt{rho\_min} \\
\midrule
\multicolumn{5}{l}{\textit{Adaptive $\rho$ rule (Boyd \S3.4.1):}} \\
\multicolumn{5}{l}{%
  $\rho \leftarrow \min(\tau\rho,\,\rho_\text{max})$ if $\|\mathbf{r}_p\| > \mu\|\mathbf{r}_d\|$;\quad
  $\rho \leftarrow \max(\rho/\tau,\,\rho_\text{min})$ if $\|\mathbf{r}_d\| > \mu\|\mathbf{r}_p\|$%
} \\
\midrule
\multicolumn{5}{l}{\textit{Residuals (explicit market clearing):}} \\
\multicolumn{5}{l}{%
  $r_{p,k}^n = D_k^n - S_k^n$\quad (demand–supply gap);\quad
  $r_{d,k}^n = \rho(S_k^n - S_k^{n-1})$\quad (change in FSP aggregate offer)%
} \\
\bottomrule
\end{tabular}
\end{table}
